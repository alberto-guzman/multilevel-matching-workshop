NoteMain takeaways
  • All worked examples in Chapters 5–7 use a single dataset, the Trends in International Mathematics and Science Study (TIMSS) 2015 Canada Grade 4 (timss), loaded from timss_df.rds.
  • The data has a multilevel structure (students nested within teachers nested within schools).

  • The treatment indicators were created from observed covariates and do not represent real interventions.

Data structure and multilevel nesting

n_students <- nrow(timss)
n_teachers <- n_distinct(timss$teacher_id)
n_schools  <- n_distinct(timss$school_id)

tibble(
  Level = c("Students", "Teachers", "Schools"),
  N     = c(n_students, n_teachers, n_schools)
) |>
  gt() |>
  tab_header(
    title    = "Multilevel structure of the analytic dataset",
    subtitle = "Data: TIMSS 2015 Grade 4 (Canada)"
  ) |>
  fmt_number(columns = N, decimals = 0) |>
  tab_style(
    style     = cell_text(weight = "bold"),
    locations = cells_title()
  )
students_per_teacher <- timss |>
  count(teacher_id, name = "n") |>
  summarise(Mean = mean(n), SD = sd(n), Min = min(n), Max = max(n)) |>
  mutate(Relationship = "Students per teacher")

teachers_per_school <- timss |>
  distinct(school_id, teacher_id) |>
  count(school_id, name = "n") |>
  summarise(Mean = mean(n), SD = sd(n), Min = min(n), Max = max(n)) |>
  mutate(Relationship = "Teachers per school")

students_per_school <- timss |>
  count(school_id, name = "n") |>
  summarise(Mean = mean(n), SD = sd(n), Min = min(n), Max = max(n)) |>
  mutate(Relationship = "Students per school")

bind_rows(students_per_teacher, teachers_per_school, students_per_school) |>
  dplyr::select(Relationship, Mean, SD, Min, Max) |>
  gt() |>
  tab_header(
    title    = "Cluster size variation",
    subtitle = "Data: TIMSS 2015 Grade 4 (Canada)"
  ) |>
  fmt_number(columns = c(Mean, SD), decimals = 1) |>
  fmt_number(columns = c(Min, Max), decimals = 0) |>
  tab_style(
    style     = cell_text(weight = "bold"),
    locations = cells_title()
  )

Intraclass correlations (ICCs)

School-level ICC

school_icc_model <- lmer(
1  math_score ~ 1 + (1 | school_id),
  data = timss
)

2school_vc     <- as.data.frame(VarCorr(school_icc_model))
3school_bw_var <- school_vc$vcov[school_vc$grp == "school_id"]
school_wi_var <- school_vc$vcov[school_vc$grp == "Residual"]
4school_icc    <- school_bw_var / (school_bw_var + school_wi_var)

school_icc
1
This fits an empty (intercept-only) random-intercept model. (1 | school_id) lets the intercept vary by school, so the model partitions the outcome variance into between-school and within-school components.
2
VarCorr() returns the estimated variance components, and as.data.frame() makes them easy to subset.
3
The grp column labels each component: the between-school variance and the residual (within-school) variance.
4
The ICC is the between-school share of the total variance. 
[1] 0.2109875

teacher_wi_var <- teacher_vc$vcov[teacher_vc$grp == "Residual"]

Three-level ICC decomposition

three_level_model <- lmer(
1  math_score ~ 1 + (1 | school_id) + (1 | school_id:teacher_id),
  data = timss
)

three_vc          <- as.data.frame(VarCorr(three_level_model))
school_var        <- three_vc$vcov[three_vc$grp == "school_id"]
teacher_var       <- three_vc$vcov[three_vc$grp == "school_id:teacher_id"]
residual_var      <- three_vc$vcov[three_vc$grp == "Residual"]
total_var         <- school_var + teacher_var + residual_var

2school_icc_3l    <- school_var   / total_var
teacher_icc_3l   <- teacher_var  / total_var

school_icc_3l
teacher_icc_3l
1
(1 | school_id:teacher_id) nests teachers within schools, so the model partitions the variance into school, teacher-within-school, and residual components.
2
Each level’s ICC is that level’s share of the total variance across the three components. 
[1] 0.2029614
[1] 0.04410661

Covariate ICCs

icc_for <- function(var) {
  fit <- lmer(reformulate("(1 | school_id)", response = var), data = timss)
  performance::icc(fit)$ICC_adjusted
}

covariate_iccs <- tibble(
  variable = c("math_score", "student_math_conf", "student_like_math",
               "student_age", "dad_edu", "mom_edu")
) |>
  mutate(icc = map_dbl(variable, icc_for))
covariate_iccs |>
  gt() |>
  cols_label(variable = "Variable", icc = "School-level ICC") |>
  tab_header(
    title    = "Covariate ICCs",
    subtitle = "Data: TIMSS 2015 Grade 4 (Canada)"
  ) |>
  fmt_number(columns = icc, decimals = 3) |>
  tab_style(
    style     = cell_text(font = "monospace"),
    locations = cells_body(columns = variable)
  ) |>
  tab_style(
    style     = cell_text(weight = "bold"),
    locations = cells_title()
  )

## Intraclass correlations (ICCs)

### School-level ICC

```{r}
#| message: false
#| warning: false
#| code-line-numbers: true

school_icc_model <- lmer(
  math_score ~ 1 + (1 | school_id),                              # <1>
  data = timss
)

school_vc     <- as.data.frame(VarCorr(school_icc_model))        # <2>
school_bw_var <- school_vc$vcov[school_vc$grp == "school_id"]    # <3>
school_wi_var <- school_vc$vcov[school_vc$grp == "Residual"]
school_icc    <- school_bw_var / (school_bw_var + school_wi_var) # <4>

school_icc
```

1. This fits an empty (intercept-only) random-intercept model. `(1 | school_id)` lets the intercept vary by school, so the model partitions the outcome variance into between-school and within-school components.
2. `VarCorr()` returns the estimated variance components, and `as.data.frame()` makes them easy to subset.
3. The `grp` column labels each component: the between-school variance and the residual (within-school) variance.
4. The ICC is the between-school share of the total variance.

### Three-level ICC decomposition

```{r}
#| message: false
#| warning: false
#| code-line-numbers: true

three_level_model <- lmer(
  math_score ~ 1 + (1 | school_id) + (1 | school_id:teacher_id), # <1>
  data = timss
)

three_vc          <- as.data.frame(VarCorr(three_level_model))
school_var        <- three_vc$vcov[three_vc$grp == "school_id"]
teacher_var       <- three_vc$vcov[three_vc$grp == "school_id:teacher_id"]
residual_var      <- three_vc$vcov[three_vc$grp == "Residual"]
total_var         <- school_var + teacher_var + residual_var

school_icc_3l    <- school_var   / total_var                     # <2>
teacher_icc_3l   <- teacher_var  / total_var

school_icc_3l
teacher_icc_3l
```

1. `(1 | school_id:teacher_id)` nests teachers within schools, so the model partitions the variance into school, teacher-within-school, and residual components.
2. Each level's ICC is that level's share of the total variance across the three components.
